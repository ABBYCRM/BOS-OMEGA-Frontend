5. Policy Engine Pseudocode

ts
export function evaluateTask(
  policy: BosLocalAutomationAgentPolicy,
  task: StructuredTaskRequest,
  approval?: ApprovalToken
): PolicyEvaluationResult {
  const reasons: string[] = [];

  if (!policy.execution_policy.structured_tasks_only) {
    return block("POLICY_INVALID_STRUCTURED_TASKS_NOT_REQUIRED");
  }

  if (policy.execution_policy.arbitrary_shell_enabled) {
    return block("ARBITRARY_SHELL_ENABLED_FORBIDDEN");
  }

  if (policy.execution_policy.persistent_shell_enabled) {
    return block("PERSISTENT_SHELL_ENABLED_FORBIDDEN");
  }

  if (!task.request_signature) {
    return block("MISSING_REQUEST_SIGNATURE");
  }

  if (isExpired(task.expires_at)) {
    return block("TASK_EXPIRED");
  }

  if (isBlocklisted(task)) {
    return block("TASK_MATCHES_BLOCKLIST");
  }

  if (!pathsAllowed(task.affected_paths)) {
    return block("AFFECTED_PATH_NOT_ALLOWED");
  }

  if (!networkDestinationsAllowed(task.network_destinations)) {
    return block("NETWORK_DESTINATION_NOT_ALLOWED");
  }

  const approvalRequirement = getApprovalRequirement(task);

  if (approvalRequirement === "BLOCKED_BY_DEFAULT") {
    return block("TASK_TIER_BLOCKED_BY_DEFAULT");
  }

  if (approvalRequirement === "OPTIONAL_AFTER_USER_ENABLEMENT") {
    return allow("LOW_RISK_DIAGNOSTIC_ALLOWED");
  }

  if (!approval) {
    return requireApproval(approvalRequirement, task.requires_elevation);
  }

  if (approval.task_id !== task.task_id) {
    return block("APPROVAL_TASK_MISMATCH");
  }

  if (approval.device_id !== task.device_id) {
    return block("APPROVAL_DEVICE_MISMATCH");
  }

  if (approval.command_hash !== task.command_hash) {
    return block("APPROVAL_COMMAND_HASH_MISMATCH");
  }

  if (isExpired(approval.expires_at)) {
    return block("APPROVAL_EXPIRED");
  }

  if (!approval.approved) {
    return block("TASK_DENIED_BY_APPROVER");
  }

  if (task.requires_elevation && !approval.elevation_approved) {
    return requireApproval("STRONG_REQUIRED_PLUS_UAC", true);
  }

  if (task.requires_elevation && !policy.elevation_policy.windows_uac_required) {
    return block("ELEVATION_WITHOUT_UAC_FORBIDDEN");
  }

  return {
    decision: "ALLOW",
    reasons: ["TASK_APPROVED_AND_POLICY_VALIDATED"],
    requires_uac: task.requires_elevation,
    risk_level: task.risk_level,
    policy_version: task.policy_version
  };

  function block(reason: string): PolicyEvaluationResult {
    return {
      decision: "BLOCK",
      reasons: [reason],
      requires_uac: false,
      risk_level: task.risk_level,
      policy_version: task.policy_version
    };
  }

  function allow(reason: string): PolicyEvaluationResult {
    return {
      decision: "ALLOW",
      reasons: [reason],
      requires_uac: false,
      risk_level: task.risk_level,
      policy_version: task.policy_version
    };
  }

  function requireApproval(
    required: ApprovalRequirement,
    requiresUac: boolean
  ): PolicyEvaluationResult {
    return {
      decision:
        required === "STRONG_REQUIRED_PLUS_UAC"
          ? "REQUIRE_STRONG_APPROVAL_PLUS_UAC"
          : required === "STRONG_REQUIRED"
            ? "REQUIRE_STRONG_APPROVAL"
            : "REQUIRE_APPROVAL",
      reasons: ["APPROVAL_REQUIRED"],
      required_approval: required,
      requires_uac: requiresUac,
      risk_level: task.risk_level,
      policy_version: task.policy_version
    };
  }
}
