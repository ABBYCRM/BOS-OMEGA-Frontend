CLASSIFICATION CONTRACT
======================================================================

type FrontDoorRoute =
  | "GREETING"
  | "EMPTY"
  | "UNDER_SPECIFIED"
  | "LIKELY_NON_TASK"
  | "VALID_TASK";

type FrontDoorClassification = {
  route: FrontDoorRoute;
  confidence: number; // 0.0 - 1.0
  rationale: string;
  shouldInvokeBosEngine: boolean;
};

======================================================================
ROUTING RULES
======================================================================

1. EMPTY
- Input is blank / whitespace only
- shouldInvokeBosEngine = false

2. GREETING
- Pure greeting / small talk starter
- Examples: hello, hi, hey, good morning
- shouldInvokeBosEngine = false

3. UNDER_SPECIFIED
- Input appears task-related but lacks enough context
- Examples:
  - "review"
  - "fix this"
  - "analyze it"
  - "thoughts?"
- shouldInvokeBosEngine = false
- Return examples and request context

4. LIKELY_NON_TASK
- Input is conversational or not suitable for BOS decision engine
- shouldInvokeBosEngine = false
- Return BOS usage guidance

5. VALID_TASK
- Input is decision / review / risk / build / compare / debug / plan / analyze shaped
- shouldInvokeBosEngine = true

6. LOW_CONFIDENCE SAFETY RULE
- If classification confidence < 0.70:
  shouldInvokeBosEngine = true

Reason:
False blocking is worse than imperfect routing.
Ambiguous real work must not be trapped at the front door.

function classifyFrontDoorInput(input, context): FrontDoorClassification {
  const raw = input ?? "";
  const text = raw.trim();
  const lower = text.toLowerCase();

  if (text.length === 0) {
    return {
      route: "EMPTY",
      confidence: 1.0,
      rationale: "Input is empty.",
      shouldInvokeBosEngine: false
    };
  }

  if (isExactGreeting(lower)) {
    return {
      route: "GREETING",
      confidence: 0.98,
      rationale: "Input is a pure greeting.",
      shouldInvokeBosEngine: false
    };
  }

  const score = scoreTaskShape(lower, context);

  if (score.validTaskScore >= 0.75) {
    return {
      route: "VALID_TASK",
      confidence: score.validTaskScore,
      rationale: score.rationale,
      shouldInvokeBosEngine: true
    };
  }

  if (score.underSpecifiedScore >= 0.75) {
    return {
      route: "UNDER_SPECIFIED",
      confidence: score.underSpecifiedScore,
      rationale: "Input appears task-related but lacks sufficient object/context.",
      shouldInvokeBosEngine: false
    };
  }

  if (score.nonTaskScore >= 0.75) {
    return {
      route: "LIKELY_NON_TASK",
      confidence: score.nonTaskScore,
      rationale: "Input does not appear to request a BOS decision/review/build task.",
      shouldInvokeBosEngine: false
    };
  }

  return {
    route: "VALID_TASK",
    confidence: 0.50,
    rationale: "Ambiguous input; routing to BOS engine to avoid false blocking.",
    shouldInvokeBosEngine: true
  };
}

Reason:

- Relevant facts
- Decision criteria
- Constraints
- Risk target
- Desired output
